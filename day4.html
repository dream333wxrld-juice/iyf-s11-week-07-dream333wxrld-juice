After (Refactored)

const MINIMUM_PRICE_THRESHOLD = 10;
const BULK_DISCOUNT_THRESHOLD = 100;
const BULK_DISCOUNT_RATE = 0.9;

function calculateOrderTotal(items) {
    if (!items || items.length === 0) {
        return 0;
    }

    const eligibleItems = items.filter(
        item => item.price > MINIMUM_PRICE_THRESHOLD
    );

    let total = eligibleItems.reduce((sum, item) => {
        return sum + (item.price * item.quantity);
    }, 0);

    if (total > BULK_DISCOUNT_THRESHOLD) {
        total = total * BULK_DISCOUNT_RATE;
    }

    return total;
}

Improvements Made

  • Meaningful function and variable names (calc → calculateOrderTotal)
  • Named constants instead of magic numbers (10, 100, 0.9)
  • Used reduce() instead of manual for loop
  • Added input validation (null/empty check)
  • Used full property names (price, quantity) instead of abbreviations
